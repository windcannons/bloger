## WebSocket 实时通讯

```javascript
onMounted(() => {
  init()
})

let socket = ref()
const init = () => {
  if (typeof (WebSocket) === "undefined") {
    alert("您的浏览器不支持 socket")
  } else {
    // 实例化 socket
    socket.value = new WebSocket("ws://skfs2002.gnway.cc/yada/websocket/Drying_tank")
    // 监听 socket 连接
    socket.value.onopen = opens
    // 监听 socket 错误信息
    socket.value.onerror = error
    // 监听 socket 消息
    socket.value.onmessage = getMessage
  }
}

// WebSocket 连接成功时的回调函数
function opens() {
  console.log("socket 连接成功")
}

// WebSocket 连接出错时的回调函数
function error() {
  console.log("连接错误")
}

// 当收到来自服务器的消息时的回调函数
function getMessage(msg) {
  // msg.data 是接收到的消息数据
  console.log('接收信息', msg.data)
}

// 向服务器发送消息的函数
function sendMessage(message) {
  if (socket.value && socket.value.readyState === WebSocket.OPEN) {
    socket.value.send(message)
  }
}
```

## Element UI 上传自动添加水印

```vue
<template>
  <el-upload
    :action="action"
    :data="fileObj"
    :file-list="fileList"
    :on-preview="handlePictureCardPreview"
    :on-remove="handleRemove"
    :on-success="handleSuccess"
    list-type="picture-card"
    :before-upload="beforeUpload">
    <i class="el-icon-plus" />
  </el-upload>
</template>

<script>
// 导入水印图片
import watermarkImage from '@/assets/logo.png'

export default {
  methods: {
    // 附件上传前触发
    beforeUpload(file) {
      // 添加水印
      var fileType = file.type
      // includes() 方法用于判断字符串是否包含指定的子字符串
      if (fileType.includes('image')) {
        var uid = file.uid
        return new Promise((resolve, reject) => {
          this.transformFile(file).then(res => {
            // 转换之后 file 的 uid 丢失
            res.uid = uid
            resolve(res)
          }).catch(error => {
            reject(error)
          })
        })
      }
    },

    // 转换文件，添加水印
    transformFile(file) {
      return new Promise((resolve, reject) => {
        const reader = new FileReader()  // 创建读取文件对象
        reader.readAsDataURL(file)  // 调用方法转 base64
        reader.onload = () => {  // 文件读取成功完成的触发
          const canvas = document.createElement('canvas')  // 新建 canvas 画布
          const img = new Image()  // 新建一个 img 标签
          img.src = reader.result  // 读取结果，下载图片
          img.onload = () => {  // 绑定事件，图片下载完成
            const ratio = window.devicePixelRatio || 1  // 获取屏幕的像素比
            const width = img.width  // 图片的宽度
            const height = img.height  // 图片的高度

            canvas.width = width * ratio  // 设置 canvas 宽度
            canvas.height = height * ratio  // 设置 canvas 高度
            const ctx = canvas.getContext('2d')  // 获取画笔对象
            ctx.scale(ratio, ratio)  // 缩放 ctx 以适配高分辨率

            // 绘制原始图片
            ctx.drawImage(img, 0, 0, width, height)

            // 创建水印图片
            const watermark = new Image()
            watermark.src = watermarkImage  // 你的水印图片路径

            watermark.onload = () => {
              // 固定水印大小，比如固定为 150x90 像素
              const watermarkHeight = height * 0.13  // 水印高度为图片高度的 13%
              const watermarkWidth = (76 / 45) * watermarkHeight  // 根据比例 76:45 计算水印宽度

              // 计算水印位置，距离右侧和底部为固定的 5px
              const padding = -5  // 距离右侧和底部的固定距离
              const x = width - watermarkWidth - padding  // 水印 X 坐标，距离右边 padding
              const y = height - watermarkHeight - padding  // 水印 Y 坐标，距离底部 padding

              // 为水印设置更高的清晰度
              ctx.globalAlpha = 1  // 设置水印透明度
              ctx.drawImage(
                watermark,
                x * ratio, y * ratio,  // 水印位置
                watermarkWidth * ratio, watermarkHeight * ratio  // 高分辨率下的宽高
              )

              // 输出高质量的 base64
              const dataUrl = canvas.toDataURL('image/jpeg', 1)  // 使用较高的压缩比例，保证质量

              // 执行下一步操作，将裁剪 base64 的图片转换为 file 文件
              resolve(this.dataURLtoFile(dataUrl, file.name))
            }

            watermark.onerror = (error) => {
              console.error('水印图片加载失败', error)  // 输出错误信息
              reject(error)  // 如果水印图片加载失败，触发 reject
            }
          }

          img.onerror = (error) => {
            console.error('图片加载失败', error)  // 输出错误信息
            reject(error)  // 如果原图片加载失败，触发 reject
          }
        }
      })
    },

    // 将裁剪 base64 的图片转换为 file 文件
    dataURLtoFile(dataurl, filename) {
      const arr = dataurl.split(',')
      const mime = arr[0].match(/:(.*?);/)[1]
      const bstr = atob(arr[1])
      let n = bstr.length
      const u8arr = new Uint8Array(n)
      while (n--) {
        u8arr[n] = bstr.charCodeAt(n)
      }
      return new File([u8arr], filename, { type: mime })
    }
  }
}
</script>
```

## Teleport：使组件可以渲染到 DOM 中的任意位置

**作用**：`Teleport` 使得组件可以渲染到 DOM 中的任何位置，脱离组件的层级结构。这对于一些需要固定位置渲染的元素，如弹出层、模态框、通知等，非常有用。

- **基本用法**：`Teleport` 可以将一个子组件或元素渲染到文档中的任意位置，而不是它的父组件的 DOM 树中。

```vue
<template>
  <Teleport to="body">
    <div class="modal">
      <p>这是一个弹窗</p>
    </div>
  </Teleport>
</template>

<style scoped>
.modal {
  position: fixed;
  top: 50%;
  left: 50%;
  transform: translate(-50%, -50%);
  background-color: white;
  padding: 20px;
  box-shadow: 0 4px 6px rgba(0, 0, 0, 0.1);
}
</style>
```

**使用场景**：对于需要脱离 DOM 层级的组件，比如弹出层、通知、模态框等，可以使用 `Teleport` 来直接渲染到 `body` 或其他 DOM 元素。

async function handleExport() {
  const table = document.getElementById('timetable-table')
  if (!table) return

  // 获取表格的 HTML 内容
  const tableContent = table.outerHTML

  // 创建 iframe
  const iframe = document.createElement('iframe')
  iframe.style.position = 'absolute'
  iframe.style.width = '0px'
  iframe.style.height = '0px'
  iframe.style.border = 'none'

  document.body.appendChild(iframe)
  const iframeDoc = iframe.contentDocument || iframe.contentWindow?.document
  if (!iframeDoc) return

  // 复制当前页面的所有样式
  let styles = ''
  Array.from(document.styleSheets).forEach((styleSheet) => {
    try {
      if (styleSheet.cssRules) {
        Array.from(styleSheet.cssRules).forEach((rule) => {
          styles += `${rule.cssText}\n`
        })
      }
    }
    catch (error) {
      console.warn(error)
    }
  })

  // 额外添加打印时隐藏 <el-icon>，确保背景色打印
  styles += `
    @media print {
      .el-icon {
        display: none !important;
      }

      thead {
        background-color: #f3f4f6 !important; /* 确保打印背景色 */
        -webkit-print-color-adjust: exact !important;
        print-color-adjust: exact !important;
      }

      th {
        background-color: #f3f4f6 !important; /* 确保表头单元格背景色 */
        color: #000 !important; /* 确保文本颜色 */
      }
    }
  `

  // 写入打印内容
  iframeDoc.open()
  iframeDoc.write(`
    <html>
      <head>
        <style>
          ${styles}
        </style>
      </head>
      <body>
        ${tableContent}
      </body>
    </html>
  `)
  iframeDoc.close()

  // 等待样式加载后再打印
  iframe.onload = () => {
    iframe.contentWindow?.print()
    document.body.removeChild(iframe)
  }
}
